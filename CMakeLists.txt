cmake_minimum_required(VERSION 3.16)
project(nlsqlc VERSION 0.1.2 LANGUAGES C)
option(NLSQL_BUILD_TESTS "Build tests" ON)
option(NLSQL_BUILD_CLI "Build CLI" ON)
option(NLSQL_BUILD_FUZZ "Build libFuzzer harness (requires Clang)" OFF)
set(NLSQL_WARNINGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Wstrict-prototypes -Wmissing-prototypes -Wformat=2 -Wundef -Werror)
add_library(nlsql STATIC src/nlsql.c)
target_compile_options(nlsql PRIVATE ${NLSQL_WARNINGS})
target_include_directories(nlsql PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(nlsql PUBLIC c_std_11)
set_target_properties(nlsql PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(nlsql_shared SHARED src/nlsql.c)
target_compile_options(nlsql_shared PRIVATE ${NLSQL_WARNINGS})
target_include_directories(nlsql_shared PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(nlsql_shared PUBLIC c_std_11)
set_target_properties(nlsql_shared PROPERTIES OUTPUT_NAME nlsql)
if(NLSQL_BUILD_CLI)
 add_executable(nlsqlc cli/nlsqlc.c)
 target_compile_options(nlsqlc PRIVATE ${NLSQL_WARNINGS})
 target_link_libraries(nlsqlc PRIVATE nlsql)
 target_link_options(nlsqlc PRIVATE -static)
 target_include_directories(nlsqlc PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
 install(TARGETS nlsqlc RUNTIME DESTINATION bin)
endif()
if(NLSQL_BUILD_TESTS)
 enable_testing()
 add_executable(nlsql-tests tests/test_core.c)
 target_compile_options(nlsql-tests PRIVATE ${NLSQL_WARNINGS} -UNDEBUG)
 target_link_libraries(nlsql-tests PRIVATE nlsql)
 add_test(NAME nlsql-core COMMAND nlsql-tests)
 add_executable(nlsql-question-tests tests/test_question.c)
 target_compile_options(nlsql-question-tests PRIVATE ${NLSQL_WARNINGS} -UNDEBUG)
 target_link_libraries(nlsql-question-tests PRIVATE nlsql)
 add_test(NAME nlsql-question COMMAND nlsql-question-tests)
endif()
if(NLSQL_BUILD_FUZZ)
 add_executable(nlsql-fuzz fuzz/fuzz_nlsql.c src/nlsql.c)
 target_compile_options(nlsql-fuzz PRIVATE ${NLSQL_WARNINGS} -fsanitize=fuzzer,address,undefined)
 target_link_options(nlsql-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
 target_include_directories(nlsql-fuzz PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
endif()
install(TARGETS nlsql nlsql_shared)
install(DIRECTORY include/ DESTINATION include)
